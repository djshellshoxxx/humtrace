cmake_minimum_required(VERSION 3.22)
project(HumTrace VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(HUMTRACE_BUILD_TESTS "Build and register tests" ON)
option(HUMTRACE_SANITIZE "Build with AddressSanitizer and UBSan" OFF)
option(HUMTRACE_BUILD_GUI "Build the Qt desktop application" ON)

add_library(humtrace_core STATIC src/spectrum.cpp src/wav.cpp src/report.cpp
            src/sha256.cpp src/analysis.cpp src/compare.cpp)
target_include_directories(humtrace_core PUBLIC include)
if(NOT MSVC)
  target_compile_options(humtrace_core PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
endif()

if(HUMTRACE_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer -g)
  add_link_options(-fsanitize=address,undefined)
endif()

add_executable(humtrace src/main.cpp)
target_link_libraries(humtrace PRIVATE humtrace_core)

if(HUMTRACE_BUILD_GUI)
  find_package(Qt6 REQUIRED COMPONENTS Widgets Test)
  set(CMAKE_AUTOMOC ON)
  add_library(humtrace_gui STATIC gui/main_window.cpp gui/timeline_widget.cpp)
  target_include_directories(humtrace_gui PUBLIC gui)
  target_link_libraries(humtrace_gui PUBLIC humtrace_core Qt6::Widgets)
  set(CMAKE_AUTORCC ON)
  set(_gui_extra packaging/humtrace.qrc)
  if(WIN32)
    list(APPEND _gui_extra packaging/humtrace.rc)
  endif()
  add_executable(humtrace-gui WIN32 gui/main.cpp ${_gui_extra})
  target_link_libraries(humtrace-gui PRIVATE humtrace_gui)
  add_executable(gui_smoke tests/gui_smoke.cpp)
  target_link_libraries(gui_smoke PRIVATE humtrace_gui Qt6::Test)
  install(TARGETS humtrace-gui humtrace RUNTIME DESTINATION bin)
  set(HUMTRACE_GUI_TEST ON)
else()
  install(TARGETS humtrace RUNTIME DESTINATION bin)
endif()

if(HUMTRACE_BUILD_TESTS)
  enable_testing()
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  if(HUMTRACE_GUI_TEST)
    add_test(NAME gui_smoke COMMAND gui_smoke)
    set_tests_properties(gui_smoke PROPERTIES ENVIRONMENT QT_QPA_PLATFORM=offscreen)
  endif()
  foreach(t test_spectrum test_wav test_report test_core fuzz_wav)
    add_executable(${t} tests/${t}.cpp)
    target_link_libraries(${t} PRIVATE humtrace_core)
    add_test(NAME ${t} COMMAND ${t})
  endforeach()
  add_test(NAME test_cli COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/test_cli.py $<TARGET_FILE:humtrace>)
endif()
