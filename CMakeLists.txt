cmake_minimum_required(VERSION 3.22)
project(HumTrace VERSION 0.2.5 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)

option(HUMTRACE_BUILD_GUI "Build the Qt 6 Widgets desktop application" ON)
option(HUMTRACE_BUILD_TESTS "Build tests" ON)

if(MSVC)
  add_compile_options(/W4 /WX)
else()
  add_compile_options(-Wall -Wextra -Wpedantic -Wconversion -Werror)
endif()

# Dependency-free core: WAV reader, DSP, tracker, compare, report, pipeline.
file(GLOB CORE_SOURCES CONFIGURE_DEPENDS ${CMAKE_SOURCE_DIR}/src/*.cpp)
list(REMOVE_ITEM CORE_SOURCES ${CMAKE_SOURCE_DIR}/src/main.cpp)
add_library(humtrace_core STATIC ${CORE_SOURCES})
target_include_directories(humtrace_core PUBLIC ${CMAKE_SOURCE_DIR}/include)

add_executable(humtrace src/main.cpp)
target_link_libraries(humtrace PRIVATE humtrace_core)
if(WIN32)
  target_link_libraries(humtrace PRIVATE shell32)
endif()
install(TARGETS humtrace RUNTIME DESTINATION bin)

# Presentation text has no Qt dependency and must remain testable in CLI-only builds.
add_library(humtrace_presentation STATIC app/presentation/findings.cpp)
target_include_directories(humtrace_presentation PUBLIC ${CMAKE_SOURCE_DIR}/app/presentation)
target_link_libraries(humtrace_presentation PUBLIC humtrace_core)

if(HUMTRACE_BUILD_TESTS)
  enable_testing()
  add_executable(app_test_presentation tests/app/test_presentation.cpp)
  target_link_libraries(app_test_presentation PRIVATE humtrace_presentation)
  add_test(NAME app_test_presentation COMMAND app_test_presentation)
  file(GLOB UNIT_TESTS CONFIGURE_DEPENDS ${CMAKE_SOURCE_DIR}/tests/test_*.cpp)
  foreach(test_src ${UNIT_TESTS})
    get_filename_component(test_name ${test_src} NAME_WE)
    add_executable(${test_name} ${test_src})
    target_link_libraries(${test_name} PRIVATE humtrace_core)
    add_test(NAME ${test_name} COMMAND ${test_name})
  endforeach()
  add_executable(settings_fuzz tests/fuzz/settings_fuzz.cpp)
  target_link_libraries(settings_fuzz PRIVATE humtrace_core)
  add_test(NAME settings_fuzz_smoke COMMAND settings_fuzz 3)
  find_package(Python3 COMPONENTS Interpreter)
  if(Python3_FOUND)
    add_test(NAME cli_paging COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/test_cli_paging.py $<TARGET_FILE:humtrace>)
    add_test(NAME detector_cli COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/test_detector_cli.py $<TARGET_FILE:humtrace>)
    add_test(NAME packaging_macos COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/test_macos_packaging.py)
    add_test(NAME packaging_macos_provenance COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/test_macos_provenance.py)
    add_test(NAME packaging_qt_sources COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/test_qt_sources.py)
    add_test(NAME packaging_licenses COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/test_package_licenses.py)
    add_test(NAME packaging_unit COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/test_packaging.py)
    add_test(NAME packaging_source COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/test_package_source.py)
    add_test(NAME source_metadata COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/test_source_metadata.py --cmake ${CMAKE_COMMAND})
    add_test(NAME cli_integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/test_cli.py $<TARGET_FILE:humtrace>)
    add_test(NAME streaming_integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/test_streaming.py $<TARGET_FILE:humtrace>)
  endif()
endif()

if(HUMTRACE_BUILD_GUI)
  find_package(Qt6 6.4 REQUIRED COMPONENTS Widgets Test)
  if(Qt6_FOUND)
    set(CMAKE_AUTOMOC ON)
    file(GLOB APP_SOURCES CONFIGURE_DEPENDS ${CMAKE_SOURCE_DIR}/app/qt/*.cpp)
    list(REMOVE_ITEM APP_SOURCES ${CMAKE_SOURCE_DIR}/app/qt/main.cpp)
    add_library(humtrace_app STATIC ${APP_SOURCES})
    target_include_directories(humtrace_app PUBLIC ${CMAKE_SOURCE_DIR}/app/qt ${CMAKE_SOURCE_DIR}/app/presentation)
    target_link_libraries(humtrace_app PUBLIC humtrace_presentation Qt6::Widgets)
    add_executable(humtrace-gui WIN32 app/qt/main.cpp)
    target_compile_definitions(humtrace-gui PRIVATE HUMTRACE_VERSION="${PROJECT_VERSION}")
    target_link_libraries(humtrace-gui PRIVATE humtrace_app)
    if(APPLE)
      set_target_properties(humtrace-gui PROPERTIES MACOSX_BUNDLE ON)
    endif()
    install(TARGETS humtrace-gui RUNTIME DESTINATION bin BUNDLE DESTINATION .)
    if(UNIX)
      install(FILES packaging/humtrace.desktop DESTINATION share/applications)
      install(FILES packaging/humtrace.svg DESTINATION share/icons/hicolor/scalable/apps)
    endif()
    if(HUMTRACE_BUILD_TESTS AND TARGET Qt6::Test)
      add_executable(gui_settings_fuzz tests/fuzz/gui_settings_fuzz.cpp)
      target_link_libraries(gui_settings_fuzz PRIVATE humtrace_app Qt6::Test)
      add_test(NAME gui_settings_fuzz_smoke COMMAND gui_settings_fuzz 3)
      set_tests_properties(gui_settings_fuzz_smoke PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
      file(GLOB APP_TESTS CONFIGURE_DEPENDS ${CMAKE_SOURCE_DIR}/tests/app/test_*.cpp)
      list(REMOVE_ITEM APP_TESTS ${CMAKE_SOURCE_DIR}/tests/app/test_presentation.cpp)
      foreach(test_src ${APP_TESTS})
        get_filename_component(test_name ${test_src} NAME_WE)
        add_executable(app_${test_name} ${test_src})
        target_link_libraries(app_${test_name} PRIVATE humtrace_app Qt6::Test)
        add_test(NAME app_${test_name} COMMAND app_${test_name})
        set_tests_properties(app_${test_name} PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
      endforeach()
    endif()
  else()
    message(FATAL_ERROR "Qt 6 is required for HUMTRACE_BUILD_GUI; use -DHUMTRACE_BUILD_GUI=OFF for the CLI")
  endif()
endif()

set(HUMTRACE_SOURCE_REVISION "" CACHE STRING "Full source revision (40 or 64 hexadecimal characters); overrides optional Git discovery")
set(HUMTRACE_SOURCE_SHA "")
if(NOT "${HUMTRACE_SOURCE_REVISION}" STREQUAL "")
  string(LENGTH "${HUMTRACE_SOURCE_REVISION}" HUMTRACE_SOURCE_REVISION_LENGTH)
  if((NOT HUMTRACE_SOURCE_REVISION_LENGTH EQUAL 40 AND NOT HUMTRACE_SOURCE_REVISION_LENGTH EQUAL 64)
      OR NOT "${HUMTRACE_SOURCE_REVISION}" MATCHES "^[0-9a-fA-F]+$")
    message(FATAL_ERROR "HUMTRACE_SOURCE_REVISION must contain exactly 40 or 64 hexadecimal characters")
  endif()
  set(HUMTRACE_SOURCE_SHA "${HUMTRACE_SOURCE_REVISION}")
else()
  find_package(Git QUIET)
  if(Git_FOUND)
    execute_process(COMMAND "${GIT_EXECUTABLE}" rev-parse --verify HEAD
                    WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}"
                    RESULT_VARIABLE HUMTRACE_GIT_RESULT
                    OUTPUT_VARIABLE HUMTRACE_SOURCE_SHA OUTPUT_STRIP_TRAILING_WHITESPACE
                    ERROR_QUIET)
    if(NOT HUMTRACE_GIT_RESULT EQUAL 0)
      set(HUMTRACE_SOURCE_SHA "")
    endif()
  endif()
endif()
if("${HUMTRACE_SOURCE_SHA}" STREQUAL "")
  set(HUMTRACE_SOURCE_SHA "unknown (source archive without Git metadata)")
endif()
file(WRITE "${CMAKE_BINARY_DIR}/SOURCE_COMMIT.txt" "${HUMTRACE_SOURCE_SHA}\n")
install(FILES ${CMAKE_BINARY_DIR}/SOURCE_COMMIT.txt DESTINATION share/doc/humtrace)
install(FILES README.md docs/user-guide/limitations.md DESTINATION share/doc/humtrace)
install(FILES docs/release/THIRD_PARTY_NOTICES.md DESTINATION share/doc/humtrace)
install(DIRECTORY schema DESTINATION share/humtrace)

set(CPACK_PACKAGE_NAME "humtrace")
set(CPACK_PACKAGE_VENDOR "Circuit Drift Labs")
set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "Audio interference measurement tool")
set(CPACK_DEBIAN_PACKAGE_MAINTAINER "Circuit Drift Labs")
set(CPACK_DEBIAN_PACKAGE_SHLIBDEPS ON)
set(CPACK_GENERATOR "TGZ")
if(UNIX AND NOT APPLE)
  list(APPEND CPACK_GENERATOR "DEB")
endif()
if(WIN32)
  set(CPACK_GENERATOR "ZIP")
endif()
include(CPack)
